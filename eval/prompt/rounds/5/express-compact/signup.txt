<a2ui>
surface("main")
$/form/name = ""
$/form/email = ""
$/form/password = ""
$/form/confirm = ""
$/form/terms = false
nameField = Input(label="Full name", name="name", value=$/form/name)
emailField = Input(label="Email", name="email", type="email", value=$/form/email, checks=[?required, ?email])
passwordField = Input(label="Password", name="password", type="password", value=$/form/password, checks=[?required, ?length(8, "At least 8 characters")])
confirmField = Input(label="Confirm password", name="confirm", type="password", value=$/form/confirm, checks=[?required, ?length(8, "At least 8 characters")])
terms = Checkbox(label="I agree to the terms", name="terms", checked=$/form/terms, checks=[?required($/form/terms, "You must agree to the terms")])
form = Form(name="signup", children=[nameField, emailField, passwordField, confirmField, terms, FormActions(submit="Create account")], on={submit: Event("create_account", {name: $/form/name, email: $/form/email, password: $/form/password})})
root = Page(title="Create account", children=[form])
</a2ui>
