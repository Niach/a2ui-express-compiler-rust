<a2ui>
surface("main")
$/form/task = ""
$/footer = "3 tasks left"
$/tasks = [{id: "t1", title: "Buy groceries", done: false}, {id: "t2", title: "Write project report", done: true}, {id: "t3", title: "Call the dentist", done: false}, {id: "t4", title: "Book flights", done: false}]

taskInput = Input(label="New task", name="task", placeholder="What needs doing?", value=$/form/task, on={submit: Event("add_task", {task: $/form/task})})
addButton = Button(label="Add", on={press: Event("add_task", {task: $/form/task})})
addBar = Stack(direction="horizontal", gap="sm", align="end", children=[taskInput, addButton])

taskBox = Checkbox(label=$title, name="done", checked=$done, on={change: Event("toggle_task", {id: $id})})
deleteBtn = Button(label="Delete", variant="ghost", icon="trash-2", on={press: Event("delete_task", {id: $id})})
item = Stack(direction="horizontal", gap="sm", align="center", justify="between", children=[taskBox, deleteBtn])
taskList = List(divided=true, children=_template($/tasks, item))

footer = Text(text=$/footer, variant="muted")

card = Card(title="My tasks", children=[addBar, taskList, Separator(), footer])
root = Page(title="To-do", children=[card])
</a2ui>
