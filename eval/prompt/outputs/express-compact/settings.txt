<a2ui>
surface("main")
$/settings/productUpdates = true
$/settings/weeklyDigest = false
$/settings/theme = "system"
$/settings/fontSize = 14
emailSection = Section(title="Email", children=[
  Switch(label="Product updates", name="productUpdates", checked=$/settings/productUpdates, on={change: Event("toggle_product_updates")}),
  Switch(label="Weekly digest", name="weeklyDigest", checked=$/settings/weeklyDigest, on={change: Event("toggle_weekly_digest")})
])
appearanceSection = Section(title="Appearance", children=[
  Select(label="Theme", name="theme", options=[{label: "Light", value: "light"}, {label: "Dark", value: "dark"}, {label: "System", value: "system"}], value=$/settings/theme),
  Slider(label="Font size", min=12, max=20, step=1, value=$/settings/fontSize, showValue=true)
])
settingsForm = Form(name="notification_settings", children=[
  emailSection,
  appearanceSection,
  FormActions(submit="Save", cancel=true, on={cancel: Event("cancel")})
], on={submit: Event("save_settings", {productUpdates: $/settings/productUpdates, weeklyDigest: $/settings/weeklyDigest, theme: $/settings/theme, fontSize: $/settings/fontSize})})
root = Page(title="Notification settings", children=[settingsForm])
</a2ui>
