<a2ui>
surface("main")
$/form/name = "Ada Lovelace"
$/form/email = "ada@example.com"
$/form/bio = "Analyst of engines"
$/form/timezone = "Europe/London"

photoRow = Stack(direction="horizontal", align="center", gap="md", children=[
  Avatar(name="Ada Lovelace", size="xl"),
  Button(label="Change photo", variant="outline", icon="ui-camera", on={press: Event("change_photo")})
])
nameField = Input(label="Display name", name="name", value=$/form/name, checks=[?required("Name is required")])
emailField = Input(label="Email", name="email", type="email", value=$/form/email, checks=[?required("Email is required"), ?email("Enter a valid email")])
bioField = Textarea(label="Bio", name="bio", rows=4, value=$/form/bio)
tzField = Select(label="Timezone", name="timezone", value=$/form/timezone, searchable=true, options=[
  {label: "Europe/London", value: "Europe/London"},
  {label: "Europe/Paris", value: "Europe/Paris"},
  {label: "Europe/Berlin", value: "Europe/Berlin"},
  {label: "America/New_York", value: "America/New_York"},
  {label: "America/Chicago", value: "America/Chicago"},
  {label: "America/Los_Angeles", value: "America/Los_Angeles"},
  {label: "Asia/Tokyo", value: "Asia/Tokyo"},
  {label: "Australia/Sydney", value: "Australia/Sydney"},
  {label: "UTC", value: "UTC"}
])
actions = FormActions(submit="Save", cancel=true, on={cancel: Event("cancel_profile")})
profileForm = Form(name="profile", children=[photoRow, nameField, emailField, bioField, tzField, actions], on={submit: Event("save_profile", {name: $/form/name, email: $/form/email, bio: $/form/bio, timezone: $/form/timezone})})
card = Card(title="Edit profile", children=[profileForm])
root = Page(title="Profile", children=[card])
</a2ui>
