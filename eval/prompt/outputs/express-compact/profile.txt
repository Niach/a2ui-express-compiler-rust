<a2ui>
surface("main")
$/form/name = "Ada Lovelace"
$/form/email = "ada@example.com"
$/form/bio = "Analyst of engines"
$/form/timezone = "Europe/London"

form = Form(name="profile", on={submit: Event("save_profile", {name: $/form/name, email: $/form/email, bio: $/form/bio, timezone: $/form/timezone})}, children=[photoRow, nameInput, emailInput, bioInput, tzSelect, actions])
root = Page(title="Edit profile", children=[form])
